==================== CURRENT HOME PAGE ====================
"use client";

import Link from "next/link";
import {
  FormEvent,
  useEffect,
  useMemo,
  useState,
} from "react";
import { useRouter } from "next/navigation";

import {
  ArrowRight,
  Bot,
  CarFront,
  Circle,
  Cog,
  Lightbulb,
  Package,
  Plus,
  Search,
  Settings,
  ShoppingCart,
  UserRound,
  Wrench,
  Zap,
} from "lucide-react";

import {
  getCurrentUser,
  type UserPublic,
} from "@/lib/api";

import {
  getAccessToken,
  removeAccessToken,
} from "@/lib/auth";

import {
  getVehicles,
  type Vehicle,
} from "@/lib/vehicles";

import {
  getBrands,
  getCategories,
  getCompatibleProducts,
  getProducts,
  type Brand,
  type Category,
  type Product,
} from "@/lib/products";

import {
  addToCart,
  CART_UPDATED_EVENT,
  getCartCount,
} from "@/lib/cart";


/* ============================================================
   HOME CATEGORY DEFINITIONS
============================================================ */

const HOME_CATEGORIES = [
  {
    label: "Brakes",
    search: "brake",
    icon: Circle,
  },
  {
    label: "Engine",
    search: "engine",
    icon: Cog,
  },
  {
    label: "Suspension",
    search: "suspension",
    icon: Wrench,
  },
  {
    label: "Electrical",
    search: "electrical",
    icon: Zap,
  },
  {
    label: "Lighting",
    search: "lighting",
    icon: Lightbulb,
  },
  {
    label: "Maintenance",
    search: "maintenance",
    icon: Settings,
  },
];


/* ============================================================
   PAGE
============================================================ */

export default function HomePage() {
  const router =
    useRouter();


  const [
    search,
    setSearch,
  ] =
    useState("");


  const [
    products,
    setProducts,
  ] =
    useState<Product[]>(
      [],
    );


  const [
    categories,
    setCategories,
  ] =
    useState<Category[]>(
      [],
    );


  const [
    brands,
    setBrands,
  ] =
    useState<Brand[]>(
      [],
    );


  const [
    activeVehicle,
    setActiveVehicle,
  ] =
    useState<Vehicle | null>(
      null,
    );


  const [
    compatibleProducts,
    setCompatibleProducts,
  ] =
    useState<Product[]>(
      [],
    );


  const [
    user,
    setUser,
  ] =
    useState<UserPublic | null>(
      null,
    );


  const [
    loadingProducts,
    setLoadingProducts,
  ] =
    useState(true);


  const [
    cartCount,
    setCartCount,
  ] =
    useState(0);


  const [
    addedProductId,
    setAddedProductId,
  ] =
    useState<string | null>(
      null,
    );


  /* ==========================================================
     CART COUNT
  ========================================================== */

  useEffect(() => {
    const updateCart =
      () => {
        setCartCount(
          getCartCount(),
        );
      };


    updateCart();


    window.addEventListener(
      CART_UPDATED_EVENT,
      updateCart,
    );


    return () => {
      window.removeEventListener(
        CART_UPDATED_EVENT,
        updateCart,
      );
    };
  }, []);


  /* ==========================================================
     PUBLIC CATALOG
  ========================================================== */

  useEffect(() => {
    let cancelled =
      false;


    async function loadCatalog() {
      try {
        setLoadingProducts(
          true,
        );


        const [
          productResult,
          categoryResult,
          brandResult,
        ] =
          await Promise.all([
            getProducts({
              sort:
                "recommended",
            }),
            getCategories(),
            getBrands(),
          ]);


        if (
          cancelled
        ) {
          return;
        }


        setProducts(
          productResult,
        );

        setCategories(
          categoryResult,
        );

        setBrands(
          brandResult,
        );
      } catch {
        /*
         * Home remains usable if catalog previews fail.
         * Shop itself handles its own retry/error state.
         */
      } finally {
        if (
          !cancelled
        ) {
          setLoadingProducts(
            false,
          );
        }
      }
    }


    void loadCatalog();


    return () => {
      cancelled =
        true;
    };
  }, []);


  /* ==========================================================
     USER + GARAGE
  ========================================================== */

  useEffect(() => {
    const token =
      getAccessToken();


    if (
      !token
    ) {
      return;
    }


    let cancelled =
      false;


    async function loadAccount() {
      try {
        const [
          currentUser,
          vehicles,
        ] =
          await Promise.all([
            getCurrentUser(
              token!,
            ),
            getVehicles(
              token!,
            ),
          ]);


        if (
          cancelled
        ) {
          return;
        }


        setUser(
          currentUser,
        );


        const active =
          vehicles.find(
            (vehicle) =>
              vehicle.is_active,
          ) ??
          null;


        setActiveVehicle(
          active,
        );


        if (
          active
        ) {
          try {
            const compatible =
              await getCompatibleProducts(
                token!,
                active.id,
              );


            if (
              !cancelled
            ) {
              setCompatibleProducts(
                compatible,
              );
            }
          } catch {
            if (
              !cancelled
            ) {
              setCompatibleProducts(
                [],
              );
            }
          }
        }
      } catch {
        if (
          !cancelled
        ) {
          removeAccessToken();

          setUser(
            null,
          );

          setActiveVehicle(
            null,
          );

          setCompatibleProducts(
            [],
          );
        }
      }
    }


    void loadAccount();


    return () => {
      cancelled =
        true;
    };
  }, []);


  /* ==========================================================
     LOOKUPS
  ========================================================== */

  const brandNames =
    useMemo(
      () =>
        new Map(
          brands.map(
            (brand) => [
              brand.id,
              brand.name,
            ],
          ),
        ),
      [
        brands,
      ],
    );


  const compatibleIds =
    useMemo(
      () =>
        new Set(
          compatibleProducts.map(
            (product) =>
              product.id,
          ),
        ),
      [
        compatibleProducts,
      ],
    );


  /* ==========================================================
     HOME PRODUCT SELECTION

     If an active vehicle exists and compatible products were
     returned, prioritize those.

     Otherwise show real recommended catalog products.
  ========================================================== */

  const featuredProducts =
    useMemo(() => {
      const source =
        compatibleProducts.length >
        0
          ? compatibleProducts
          : products;


      return source
        .filter(
          (product) =>
            product.is_active &&
            product.stock_quantity >
              0,
        )
        .slice(
          0,
          4,
        );
    }, [
      compatibleProducts,
      products,
    ]);


  /* ==========================================================
     CATEGORY LINK
  ========================================================== */

  function categoryHref(
    label: string,
    fallbackSearch: string,
  ) {
    const normalized =
      label
        .toLowerCase();


    const category =
      categories.find(
        (item) =>
          item.name
            .toLowerCase()
            .includes(
              normalized,
            ) ||
          item.slug
            .toLowerCase()
            .includes(
              normalized,
            ),
      );


    if (
      category
    ) {
      return (
        `/shop?category_id=${encodeURIComponent(
          category.id,
        )}`
      );
    }


    return (
      `/shop?search=${encodeURIComponent(
        fallbackSearch,
      )}`
    );
  }


  /* ==========================================================
     SEARCH
  ========================================================== */

  function submitSearch(
    event:
      FormEvent<HTMLFormElement>,
  ) {
    event.preventDefault();


    const value =
      search.trim();


    router.push(
      value
        ? `/shop?search=${encodeURIComponent(
            value,
          )}`
        : "/shop",
    );
  }


  /* ==========================================================
     ADD TO CART
  ========================================================== */

  function handleAddToCart(
    product: Product,
  ) {
    if (
      product.stock_quantity <=
      0
    ) {
      return;
    }


    addToCart(
      product,
    );


    setCartCount(
      getCartCount(),
    );


    setAddedProductId(
      product.id,
    );


    window.setTimeout(
      () => {
        setAddedProductId(
          (
            current,
          ) =>
            current ===
            product.id
              ? null
              : current,
        );
      },
      900,
    );
  }


  /* ==========================================================
     RENDER
  ========================================================== */

  return (
    <main className="min-h-screen bg-[#f4f6f8] text-[#101828]">
      <HomeNavbar
        search={
          search
        }
        setSearch={
          setSearch
        }
        submitSearch={
          submitSearch
        }
        cartCount={
          cartCount
        }
        user={
          user
        }
      />


      <section className="mx-auto max-w-[1320px] px-4 pb-16 pt-10 sm:px-6 lg:px-8">
        {/* ====================================================
            HERO
        ==================================================== */}

        <section className="relative overflow-hidden rounded-[18px] bg-[#071523] px-7 py-8 text-white shadow-[0_14px_35px_rgba(7,21,35,0.12)] sm:px-9 lg:min-h-[305px] lg:px-10 lg:py-10">
          <div
            className="pointer-events-none absolute inset-0 opacity-[0.055]"
            style={{
              backgroundImage:
                "linear-gradient(rgba(255,255,255,.3) 1px, transparent 1px), linear-gradient(90deg, rgba(255,255,255,.3) 1px, transparent 1px)",
              backgroundSize:
                "48px 48px",
            }}
          />


          <div className="relative z-10 grid gap-8 lg:grid-cols-[1.45fr_.75fr] lg:items-center">
            <div>
              <div className="inline-flex rounded-full border border-[#6d2934] bg-[#341721] px-3 py-1 text-[9px] font-bold uppercase tracking-[0.13em] text-[#ff8a97]">
                AI-powered auto care
              </div>


              <h1 className="mt-5 max-w-[650px] text-[38px] font-bold leading-[1.02] tracking-[-0.045em] sm:text-[46px] lg:text-[54px]">
                Know the problem.
                <br />
                Find the right part.
              </h1>


              <p className="mt-5 max-w-[690px] text-[13px] leading-6 text-[#a9b6c3] sm:text-[14px]">
                Add your vehicle, describe the symptoms and let Vehnexa connect possible causes to compatible parts in the marketplace.
              </p>


              <div className="mt-6 flex flex-wrap gap-3">
                <Link
                  href="/ai-mechanic"
                  className="inline-flex h-11 items-center justify-center gap-2 rounded-[6px] bg-[#ef3d43] px-6 text-[12px] font-semibold text-white transition hover:bg-[#db3037]"
                >
                  <Bot className="h-4 w-4" />

                  Diagnose my car
                </Link>


                <Link
                  href="/shop"
                  className="inline-flex h-11 items-center justify-center gap-2 rounded-[6px] border border-[#45586a] px-6 text-[12px] font-semibold text-white transition hover:bg-white/[0.05]"
                >
                  Browse parts

                  <ArrowRight className="h-4 w-4" />
                </Link>
              </div>
            </div>


            <VehicleProfileCard
              vehicle={
                activeVehicle
              }
            />
          </div>
        </section>


        {/* ====================================================
            CATEGORIES
        ==================================================== */}

        <section className="mt-7">
          <div className="flex items-end justify-between gap-4">
            <h2 className="text-[20px] font-bold tracking-[-0.02em]">
              Shop by category
            </h2>


            <Link
              href="/shop"
              className="text-[11px] font-semibold text-[#667085] transition hover:text-[#e31b2d]"
            >
              View all parts
            </Link>
          </div>


          <div className="mt-4 grid grid-cols-2 gap-3 md:grid-cols-3 xl:grid-cols-6">
            {HOME_CATEGORIES.map(
              (
                category,
              ) => {
                const Icon =
                  category.icon;


                return (
                  <Link
                    key={
                      category.label
                    }
                    href={
                      categoryHref(
                        category.label,
                        category.search,
                      )
                    }
                    className="group flex min-h-[70px] items-center gap-3 rounded-[8px] border border-[#dfe3e8] bg-white px-4 transition hover:-translate-y-[1px] hover:border-[#cfd5dc] hover:shadow-[0_7px_18px_rgba(15,23,42,0.05)]"
                  >
                    <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-[#f0f3f6] text-[#667085] transition group-hover:bg-[#fff0f1] group-hover:text-[#e31b2d]">
                      <Icon className="h-4 w-4" />
                    </span>


                    <span className="text-[12px] font-semibold text-[#344054]">
                      {
                        category.label
                      }
                    </span>
                  </Link>
                );
              },
            )}
          </div>
        </section>


        {/* ====================================================
            POPULAR PRODUCTS
        ==================================================== */}

        <section className="mt-8">
          <div className="flex flex-wrap items-end justify-between gap-4">
            <div>
              <h2 className="text-[20px] font-bold tracking-[-0.02em]">
                Popular parts for your garage
              </h2>


              <p className="mt-1 text-[11px] text-[#667085]">
                {activeVehicle
                  ? `Showing parts for your ${activeVehicle.year} ${activeVehicle.make} ${activeVehicle.model}.`
                  : "Popular items from the Vehnexa marketplace."}
              </p>
            </div>


            {activeVehicle && (
              <Link
                href={
                  `/shop?vehicleId=${encodeURIComponent(
                    activeVehicle.id,
                  )}&fits=true`
                }
                className="text-[11px] font-semibold text-[#e31b2d]"
              >
                Browse compatible parts ?
              </Link>
            )}
          </div>


          {loadingProducts ? (
            <FeaturedProductsLoading />
          ) : featuredProducts.length >
            0 ? (
            <div className="mt-5 grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
              {featuredProducts.map(
                (
                  product,
                  index,
                ) => (
                  <HomeProductCard
                    key={
                      product.id
                    }
                    product={
                      product
                    }
                    brandName={
                      brandNames.get(
                        product.brand_id,
                      ) ??
                      "Vehnexa"
                    }
                    compatible={
                      compatibleIds.has(
                        product.id,
                      )
                    }
                    index={
                      index
                    }
                    added={
                      addedProductId ===
                      product.id
                    }
                    onAdd={() =>
                      handleAddToCart(
                        product,
                      )
                    }
                  />
                ),
              )}
            </div>
          ) : (
            <div className="mt-5 rounded-[8px] border border-[#dfe3e8] bg-white px-6 py-10 text-center">
              <Package className="mx-auto h-8 w-8 text-[#98a2b3]" />

              <p className="mt-3 text-[13px] font-semibold text-[#344054]">
                Marketplace products are temporarily unavailable.
              </p>


              <Link
                href="/shop"
                className="mt-4 inline-flex text-[11px] font-semibold text-[#e31b2d]"
              >
                Open marketplace
              </Link>
            </div>
          )}
        </section>
      </section>


      <footer className="border-t border-[#182a3d] bg-[#071523]">
        <div className="mx-auto flex max-w-[1320px] flex-col gap-2 px-4 py-5 text-[9px] uppercase tracking-[0.12em] text-[#8393a3] sm:flex-row sm:items-center sm:justify-between sm:px-6 lg:px-8">
          <span>
            Vehnexa
          </span>

          <span>
            AI-powered automotive marketplace
          </span>
        </div>
      </footer>
    </main>
  );
}


/* ============================================================
   NAVBAR
============================================================ */

function HomeNavbar({
  search,
  setSearch,
  submitSearch,
  cartCount,
  user,
}: {
  search:
    string;

  setSearch:
    (
      value:
        string,
    ) =>
      void;

  submitSearch:
    (
      event:
        FormEvent<HTMLFormElement>,
    ) =>
      void;

  cartCount:
    number;

  user:
    UserPublic | null;
}) {
  const initials =
    user
      ? `${user.first_name?.[0] ?? ""}${user.last_name?.[0] ?? ""}`
          .toUpperCase()
      : "";


  return (
    <header className="border-b border-[#e4e7ec] bg-white">
      <div className="mx-auto flex h-[72px] max-w-[1320px] items-center gap-6 px-4 sm:px-6 lg:px-8">
        <Link
          href="/"
          className="flex shrink-0 items-center gap-2.5"
        >
          <VehnexaMark />

          <span className="text-[16px] font-bold tracking-[-0.02em] text-[#17202a]">
            Vehnexa
          </span>
        </Link>


        <nav className="ml-4 hidden h-full items-center gap-8 lg:flex">
          <TopLink
            href="/"
            label="Home"
            active
          />

          <TopLink
            href="/shop"
            label="Shop"
          />

          <TopLink
            href="/account/garage"
            label="My Garage"
          />

          <TopLink
            href="/ai-mechanic"
            label="AI Mechanic"
          />

          <TopLink
            href="/orders"
            label="Orders"
          />
        </nav>


        <div className="ml-auto flex items-center gap-2">
          <form
            onSubmit={
              submitSearch
            }
            className="relative hidden lg:block"
          >
            <Search className="pointer-events-none absolute left-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-[#98a2b3]" />

            <input
              value={
                search
              }
              onChange={(
                event,
              ) =>
                setSearch(
                  event.target.value,
                )
              }
              placeholder="Search parts..."
              className="h-10 w-[205px] rounded-full border border-[#d8dde3] bg-[#f7f8fa] pl-9 pr-4 text-[11px] text-[#344054] outline-none transition placeholder:text-[#98a2b3] focus:border-[#b9c1ca] focus:bg-white"
            />
          </form>


          <Link
            href="/cart"
            className="relative flex h-9 w-9 items-center justify-center rounded-full text-[#667085] transition hover:bg-[#f2f4f7]"
            aria-label="Cart"
          >
            <ShoppingCart className="h-[17px] w-[17px]" />

            {cartCount >
              0 && (
              <span className="absolute -right-0.5 -top-0.5 flex min-h-[16px] min-w-[16px] items-center justify-center rounded-full bg-[#e31b2d] px-1 text-[8px] font-bold text-white">
                {cartCount >
                99
                  ? "99+"
                  : cartCount}
              </span>
            )}
          </Link>


          <Link
            href="/account"
            className="flex h-9 w-9 items-center justify-center rounded-full bg-[#071523] text-[10px] font-bold text-white"
            aria-label="Account"
          >
            {initials ? (
              initials
            ) : (
              <UserRound className="h-[16px] w-[16px]" />
            )}
          </Link>
        </div>
      </div>
    </header>
  );
}


/* ============================================================
   VEHICLE PROFILE
============================================================ */

function VehicleProfileCard({
  vehicle,
}: {
  vehicle:
    Vehicle | null;
}) {
  if (
    !vehicle
  ) {
    return (
      <div className="rounded-[14px] border border-white/[0.06] bg-[#101d30] p-5">
        <div className="inline-flex rounded-full bg-[#172f2a] px-3 py-1 text-[8px] font-bold uppercase tracking-[0.1em] text-[#62d39c]">
          Vehicle profile
        </div>


        <div className="flex min-h-[150px] flex-col items-center justify-center text-center">
          <CarFront className="h-16 w-16 text-[#53687c]" />

          <p className="mt-4 text-[12px] font-semibold text-white">
            Add your vehicle
          </p>

          <p className="mt-1 max-w-[230px] text-[10px] leading-4 text-[#8294a5]">
            Save a vehicle to unlock compatibility-aware shopping and AI diagnostics.
          </p>


          <Link
            href="/account/garage/add"
            className="mt-4 inline-flex items-center gap-1 text-[10px] font-semibold text-[#ff6675]"
          >
            Add vehicle

            <Plus className="h-3 w-3" />
          </Link>
        </div>
      </div>
    );
  }


  return (
    <div className="rounded-[14px] border border-white/[0.06] bg-[#101d30] p-5">
      <div className="flex items-center justify-between gap-3">
        <div className="inline-flex rounded-full bg-[#123126] px-3 py-1 text-[8px] font-bold uppercase tracking-[0.1em] text-[#63d79e]">
          Vehicle profile active
        </div>


        <Link
          href="/account/garage"
          className="text-[8px] font-semibold uppercase tracking-[0.1em] text-[#71869a]"
        >
          Manage
        </Link>
      </div>


      <div className="mt-4 flex min-h-[145px] items-center justify-center overflow-hidden rounded-[10px] bg-[#0d1929]">
        {vehicle.image_url ? (
          <img
            src={
              vehicle.image_url
            }
            alt={
              `${vehicle.year} ${vehicle.make} ${vehicle.model}`
            }
            className="h-[145px] w-full object-contain p-3"
          />
        ) : (
          <CarFront className="h-20 w-20 text-[#566b80]" />
        )}
      </div>


      <div className="mt-4">
        <p className="text-[14px] font-semibold text-white">
          {vehicle.year}{" "}
          {vehicle.make}{" "}
          {vehicle.model}
        </p>


        <p className="mt-1 text-[10px] text-[#8395a6]">
          {vehicle.engine}{" "}
          ?{" "}
          {vehicle.transmission}
        </p>
      </div>
    </div>
  );
}


/* ============================================================
   PRODUCT CARD
============================================================ */

function HomeProductCard({
  product,
  brandName,
  compatible,
  index,
  added,
  onAdd,
}: {
  product:
    Product;

  brandName:
    string;

  compatible:
    boolean;

  index:
    number;

  added:
    boolean;

  onAdd:
    () =>
      void;
}) {
  const effectivePrice =
    product.sale_price ??
    product.price;


  let badge:
    string | null =
    null;


  if (
    compatible
  ) {
    badge =
      "Fits your vehicle";
  } else if (
    index ===
    1
  ) {
    badge =
      "Popular";
  } else if (
    product.stock_quantity <=
    8
  ) {
    badge =
      "Low stock";
  }


  const image =
    product.images?.[
      0
    ] ??
    null;


  return (
    <article className="overflow-hidden rounded-[9px] border border-[#dfe3e8] bg-white transition hover:-translate-y-[1px] hover:shadow-[0_10px_25px_rgba(15,23,42,0.06)]">
      <div className="relative m-3 flex h-[170px] items-center justify-center overflow-hidden rounded-[7px] bg-[#f1f3f6]">
        {badge && (
          <span className="absolute left-2 top-2 z-10 rounded-full bg-[#ffe9eb] px-2.5 py-1 text-[8px] font-bold text-[#e43b48]">
            {
              badge
            }
          </span>
        )}


        {image ? (
          <img
            src={
              image
            }
            alt={
              product.name
            }
            className="h-full w-full object-contain p-4"
          />
        ) : (
          <Package className="h-12 w-12 text-[#c2cbd4]" />
        )}
      </div>


      <div className="px-4 pb-4">
        <p className="text-[8px] font-bold uppercase tracking-[0.11em] text-[#667085]">
          {
            brandName
          }
        </p>


        <Link

async function readResponse(
  response: Response,
) {
  if (response.status === 204) {
    return null;
  }

  const contentType =
    response.headers.get(
      "content-type",
    );

  if (
    contentType?.includes(
      "application/json",
    )
  ) {
    return response.json();
  }

  const text =
    await response.text();

  return {
    detail:
      text ||
      "Unexpected server response.",
  };
}


/* ============================================================
   PUBLIC JSON REQUEST
============================================================ */

async function getPublicJson<T>(
  url: string,
  fallbackMessage: string,
): Promise<T> {
  const response = await fetch(
    url,
    {
      method: "GET",
      headers: {
        Accept: "application/json",
      },
      cache: "no-store",
    },
  );

  const data =
    await readResponse(
      response,
    );

  if (!response.ok) {
    throw new ApiError(
      data?.detail ??
        fallbackMessage,
      response.status,
    );
  }

  return data as T;
}

    );

    params.set(
      "subcategory_slug",
      query.subcategory_slug,
    );
  }

  if (
    query.min_price !==
    undefined
  ) {
    params.set(
      "min_price",
      String(
        query.min_price,
      ),
    );
  }

  if (
    query.max_price !==
    undefined
  ) {
    params.set(
      "max_price",
      String(
        query.max_price,

    );
  }

  if (
    query.in_stock !==

      query.sort,
    );

    params.toString();

  const url = queryString
    ? `${API_BASE_URL}/api/products?${queryString}`
    : `${API_BASE_URL}/api/products`;

import type {
  Product,
} from "@/lib/products";


const CART_STORAGE_KEY =
  "vehnexa_cart";

export const CART_UPDATED_EVENT =
  "vehnexa-cart-updated";


/* ============================================================
   CART ITEM
============================================================ */

export type CartItem = {
  product_id: string;
  name: string;
  slug: string;
  sku: string;
  brand_id: string;
  category_id: string;
  image_url: string | null;

  price: number;
  sale_price: number | null;

  stock_quantity: number;
  quantity: number;
};


/* ============================================================
   BROWSER GUARD
============================================================ */

function canUseStorage(): boolean {
  return (
    typeof window !==
    "undefined"
  );
}


/* ============================================================
   READ CART
============================================================ */

export function getCartItems(): CartItem[] {
  if (!canUseStorage()) {
    return [];
  }

  try {
    const raw =
      window.localStorage.getItem(
        CART_STORAGE_KEY,
      );

    if (!raw) {
      return [];
    }

    const parsed: unknown =
      JSON.parse(raw);

    if (!Array.isArray(parsed)) {
      return [];
    }

    return parsed.filter(
      (
        item,
      ): item is CartItem => {
        if (
          typeof item !==
            "object" ||
          item === null
        ) {
          return false;
        }

        const candidate =
          item as Partial<CartItem>;

        return (
          typeof candidate.product_id ===
            "string" &&
          typeof candidate.name ===
            "string" &&
          typeof candidate.quantity ===
            "number" &&
          typeof candidate.stock_quantity ===
            "number"
        );
      },
    );
  } catch {
    return [];
  }
}


/* ============================================================
   WRITE CART
============================================================ */

function saveCartItems(
  items: CartItem[],
): void {
  if (!canUseStorage()) {
    return;
  }

  window.localStorage.setItem(
    CART_STORAGE_KEY,
    JSON.stringify(items),
  );

  window.dispatchEvent(
    new Event(
      CART_UPDATED_EVENT,
    ),
  );
}


/* ============================================================
   PRODUCT → CART ITEM
============================================================ */

function productToCartItem(
  product: Product,
  quantity: number,
): CartItem {
  return {
    product_id: product.id,
    name: product.name,
    slug: product.slug,
    sku: product.sku,

    brand_id:
      product.brand_id,

    category_id:
      product.category_id,

    image_url:
      product.images[0] ??
      null,

    price: product.price,

    sale_price:
      product.sale_price,

    stock_quantity:
      product.stock_quantity,

    quantity,
  };
}


/* ============================================================
   ADD TO CART
============================================================ */

export function addToCart(
  product: Product,
  quantity = 1,
): CartItem[] {
  const items =
    getCartItems();

  if (
    product.stock_quantity <=
    0
  ) {
    return items;
  }

  const safeRequestedQuantity =
    Math.max(
      1,
      Math.floor(quantity),
    );

  const existingIndex =
    items.findIndex(
      (item) =>
        item.product_id ===
        product.id,
    );

  if (existingIndex >= 0) {
    const existing =
      items[existingIndex];

    const updatedQuantity =
      Math.min(
        existing.quantity +
          safeRequestedQuantity,
        product.stock_quantity,
      );

    items[existingIndex] = {
      ...existing,

      name: product.name,
      slug: product.slug,
      sku: product.sku,

      brand_id:
        product.brand_id,

      category_id:
        product.category_id,

      image_url:
        product.images[0] ??
        null,

      price:
        product.price,

      sale_price:
        product.sale_price,

      stock_quantity:
        product.stock_quantity,

      quantity:
        updatedQuantity,
    };
  } else {
    const initialQuantity =
      Math.min(
        safeRequestedQuantity,
        product.stock_quantity,
      );

    items.push(
      productToCartItem(
        product,
        initialQuantity,
      ),
    );
  }

  saveCartItems(items);

  return items;
}


/* ============================================================
   UPDATE QUANTITY
============================================================ */

export function updateCartQuantity(
  productId: string,
  quantity: number,
): CartItem[] {
  const items =
    getCartItems();

  const index =
    items.findIndex(
      (item) =>
        item.product_id ===
        productId,
    );

  if (index < 0) {
    return items;
  }

  if (quantity <= 0) {
    const filtered =
      items.filter(
        (item) =>
          item.product_id !==
          productId,
      );

    saveCartItems(filtered);

    return filtered;
  }

  const item =
    items[index];

  const safeQuantity =
    Math.min(
      Math.max(
        1,
        Math.floor(quantity),
      ),
      item.stock_quantity,
    );

  items[index] = {
    ...item,
    quantity:
      safeQuantity,
  };

  saveCartItems(items);

  return items;
}


/* ============================================================
   REMOVE ITEM
============================================================ */

export function removeFromCart(
  productId: string,
): CartItem[] {
  const items =
    getCartItems().filter(
      (item) =>
        item.product_id !==
        productId,
    );

  saveCartItems(items);

  return items;
}


/* ============================================================
   CLEAR CART
============================================================ */

export function clearCart(): void {
  if (!canUseStorage()) {
    return;
  }

  window.localStorage.removeItem(
    CART_STORAGE_KEY,
  );

  window.dispatchEvent(
    new Event(
      CART_UPDATED_EVENT,
    ),
  );
}


/* ============================================================
   CART COUNT

   Counts quantities, not just distinct products.

   Example:
   Brake Pads × 2
   Oil Filter × 1

   count = 3
============================================================ */

export function getCartCount(): number {
  return getCartItems().reduce(
    (
      total,
      item,
    ) =>
      total +
      item.quantity,
    0,
  );
}


/* ============================================================
   CART SUBTOTAL
============================================================ */

export function getCartSubtotal(): number {
  return getCartItems().reduce(
    (
      total,
      item,
    ) => {
      const unitPrice =
        item.sale_price ??
        item.price;

      return (
        total +
        unitPrice *
          item.quantity
      );
    },
    0,
  );
}

): Promise<TokenResponse> {
  const response = await fetch(`${API_BASE_URL}/api/auth/login`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(payload),
  });

  const data = await readResponse(response);

  if (!response.ok) {
    throw new ApiError(
      data?.detail ?? "Unable to sign in.",
      response.status,
    );
  }

  return data;
}

export async function getCurrentUser(
  token: string,
): Promise<UserPublic> {
  const response = await fetch(`${API_BASE_URL}/api/auth/me`, {
    headers: {
      Authorization: `Bearer ${token}`,
    },
  });

  const data = await readResponse(response);

  if (!response.ok) {
    throw new ApiError(
      data?.detail ?? "Unable to load your account.",
      response.status,
    );
  }

  return data;
}

const TOKEN_KEY = "partpilot_access_token";

export function saveAccessToken(token: string) {
  if (typeof window === "undefined") {
    return;
  }

  localStorage.setItem(TOKEN_KEY, token);
}

export function getAccessToken(): string | null {
  if (typeof window === "undefined") {
    return null;
  }

  return localStorage.getItem(TOKEN_KEY);
}

export function removeAccessToken() {
  if (typeof window === "undefined") {
    return;
  }

  localStorage.removeItem(TOKEN_KEY);
}

export function isAuthenticated(): boolean {
  return Boolean(getAccessToken());
}
